     '''
        import matplotlib as mpl
        import matplotlib.pyplot as plt
        import numpy as np
        from mpl_toolkits.mplot3d.art3d import Poly3DCollection
        from matplotlib.widgets import Slider, Button, CheckButtons

        mpl.rcParams['lines.antialiased'] = True
        mpl.rcParams['patch.antialiased'] = True

        # 默认参数需要遵守--------------
        mpl.rcParams["axes3d.mouserotationstyle"] = "azel"
        plt.rcParams.update({
            "font.family": "SimHei",
            "mathtext.fontset": "cm",
            "axes.unicode_minus": False,
            "font.size": 11
        })
        # ------------------------------------

        MORANDI = {
            'plane_alpha': '#BFD4E6',
            'plane_beta':  '#B8DCCF',
            'plane_gamma': '#AFC4E8',
            'line_main':   '#2F3A40',
            'line_aux':    '#AEB9BF',
            'bg':          '#F7FAFA',
            'track':       '#E8EEEE',
        }
        # 色系计算

        def calc_intersection(theta_deg):
            """计算动平面绕横轴旋转切割单位正方体截面。"""
            theta = np.radians(theta_deg)
            t = np.tan(theta)

            if t <= 1:
                verts = [(0, 0, 0), (1, 0, 0), (1, 1, t), (0, 1, t)]
                area = np.sqrt(1 + t ** 2)
            else:
                verts = [(0, 0, 0), (1, 0, 0), (1, 1 / t, 1), (0, 1 / t, 1)]
                area = np.sqrt((1 / t) ** 2 + 1)

            return verts, area


        fig = plt.figure(figsize=(10, 5), facecolor=MORANDI['bg'])

        # [左图]: 3D 空间
        ax1 = fig.add_axes([0.02, 0.22, 0.45, 0.70], projection='3d', proj_type='ortho')
        ax1.set_axis_off()
        ax1.set_facecolor(MORANDI['bg'])
        ax1.view_init(elev=20, azim=-45)

        ax1.set_xlim(0, 1)
        ax1.set_ylim(0, 1)
        ax1.set_zlim(0, 1)

        ax1.text2D(
            0.5, 0.95,
            "空间几何: 动平面截面演化",
            transform=ax1.transAxes,
            ha='center',
            fontsize=13,
            color=MORANDI['line_main']
        )

        # [右图]: 2D 曲线
        ax2 = fig.add_axes([0.55, 0.25, 0.38, 0.60])
        ax2.set_facecolor(MORANDI['bg'])
        ax2.spines['top'].set_visible(False)
        ax2.spines['right'].set_visible(False)
        ax2.spines['left'].set_color(MORANDI['line_aux'])
        ax2.spines['bottom'].set_color(MORANDI['line_aux'])
        ax2.tick_params(colors=MORANDI['line_main'])

        ax2.set_title("截面积曲线 $S(\\theta)$", pad=15, fontsize=13, color=MORANDI['line_main'])
        ax2.set_xlabel(r"$\theta\ (^{\circ})$", color=MORANDI['line_main'])

        # 单位正方体边线
        cube_lines = []
        for p in (0, 1):
            cube_lines.append(ax1.plot([0, 1], [p, p], [0, 0], color=MORANDI['line_aux'], lw=0.8, ls='--')[0])
            cube_lines.append(ax1.plot([0, 1], [p, p], [1, 1], color=MORANDI['line_aux'], lw=0.8, ls='--')[0])
            cube_lines.append(ax1.plot([p, p], [0, 1], [0, 0], color=MORANDI['line_aux'], lw=0.8, ls='--')[0])
            cube_lines.append(ax1.plot([p, p], [0, 1], [1, 1], color=MORANDI['line_aux'], lw=0.8, ls='--')[0])
            cube_lines.append(ax1.plot([p, p], [0, 0], [0, 1], color=MORANDI['line_aux'], lw=0.8, ls='--')[0])
            cube_lines.append(ax1.plot([p, p], [1, 1], [0, 1], color=MORANDI['line_aux'], lw=0.8, ls='--')[0])

        init_verts, init_area = calc_intersection(30)

        poly3d = Poly3DCollection(
            [init_verts],
            facecolors=MORANDI['plane_alpha'],
            edgecolors=MORANDI['line_main'],
            linewidths=1.2,
            alpha=0.4
        )
        ax1.add_collection3d(poly3d)

        vertex_label_artists = []


        def draw_vertex_labels(verts):
            """刷新截面顶点标签，避免使用 global 污染模块变量。"""
            for artist in vertex_label_artists:
                artist.remove()
            vertex_label_artists.clear()

            labels = [r"$A$", r"$B$", r"$C$", r"$D$"]
            for label, (x, y, z) in zip(labels, verts):
                dx = -0.04 if x == 0 else 0.04
                dy = -0.04 if y == 0 else 0.04
                dz = -0.04 if z == 0 else 0.04

                artist = ax1.text(
                    x + dx, y + dy, z + dz,
                    label,
                    fontsize=13,
                    color=MORANDI['line_main'],
                    ha='center',
                    va='center'
                )
                vertex_label_artists.append(artist)


        draw_vertex_labels(init_verts)

        thetas = np.linspace(0, 89.9, 200)
        areas = np.array([calc_intersection(th)[1] for th in thetas])

        ax2.plot(thetas, areas, color=MORANDI['plane_gamma'], lw=1.2)
        dot2d, = ax2.plot([30], [init_area], 'o', color=MORANDI['line_main'], ms=5)

        fill_state = {
            'artist': ax2.fill_between(
                thetas, areas,
                where=(thetas <= 30),
                color=MORANDI['plane_gamma'],
                alpha=0.15
            )
        }

        # 布局交互 UI 组件
        ax_slider = fig.add_axes([0.15, 0.08, 0.40, 0.02])
        ax_chk = fig.add_axes([0.62, 0.05, 0.15, 0.08])
        ax_btn = fig.add_axes([0.82, 0.06, 0.08, 0.05])

        ax_slider.set_facecolor(MORANDI['bg'])
        ax_chk.set_facecolor(MORANDI['bg'])
        ax_btn.set_facecolor(MORANDI['bg'])

        # 滑块配置：不要 ax_slider.set_axis_off()，否则容易影响 Slider 自己的文本/handle
        slider = Slider(
            ax=ax_slider,
            label='旋转角 $\\theta$',
            valmin=0,
            valmax=89.9,
            valinit=30,
            valfmt='%1.1f°',
            color=MORANDI['plane_beta'],
            track_color=MORANDI['track'],
            initcolor='none',
            handle_style={
                'facecolor': MORANDI['line_main'],
                'edgecolor': 'white',
                'size': 8,
            }
        )

        for spine in ax_slider.spines.values():
            spine.set_visible(False)
        ax_slider.set_xticks([])
        ax_slider.set_yticks([])

        slider.label.set_color(MORANDI['line_main'])
        slider.valtext.set_color(MORANDI['line_main'])


        chk = CheckButtons(
            ax=ax_chk,
            labels=['辅助框线'],
            actives=[True],
            label_props={
                'color': [MORANDI['line_main']],
            },
            frame_props={
                'edgecolor': [MORANDI['line_aux']],
                'facecolor': ['white'],
                'linewidth': [0.8],
                's': [36],
            },
            check_props={
                'facecolor': [MORANDI['line_main']],
                'linewidth': [1.2],
                's': [36],
            }
        )

        for spine in ax_chk.spines.values():
            spine.set_visible(False)
        ax_chk.set_xticks([])
        ax_chk.set_yticks([])

        # 重置按钮配置
        btn = Button(ax_btn, '重置', color='white', hovercolor='#F1F6F6')
        for spine in ax_btn.spines.values():
            spine.set_color(MORANDI['line_aux'])
            spine.set_linewidth(0.8)
        btn.label.set_color(MORANDI['line_main'])


        def update(val):
            theta = slider.val
            verts, area = calc_intersection(theta)

            poly3d.set_verts([verts])
            draw_vertex_labels(verts)
            dot2d.set_data([theta], [area])

            fill_state['artist'].remove()
            fill_state['artist'] = ax2.fill_between(
                thetas, areas,
                where=(thetas <= theta),
                color=MORANDI['plane_gamma'],
                alpha=0.15
            )

            fig.canvas.draw_idle()


        def toggle_visibility(label):
            visible = chk.get_status()[0]
            for line in cube_lines:
                line.set_visible(visible)
            fig.canvas.draw_idle()


        slider.on_changed(update)
        chk.on_clicked(toggle_visibility)
        btn.on_clicked(lambda event: slider.reset())

        plt.show()
    '''
