        import matplotlib as mpl

        from matplotlib.widgets import Slider, Button, CheckButtons
        
        mpl.rcParams['lines.antialiased'] = mpl.rcParams['patch.antialiased'] = True
        mpl.rcParams["axes3d.mouserotationstyle"] = "azel"
        plt.rcParams.update({"font.family": "SimHei", "mathtext.fontset": "cm",
                             "axes.unicode_minus": False, "font.size": 11})
        
        MORANDI = {
            'plane_alpha': '#BFD4E6', 'plane_beta': '#B8DCCF', 'plane_gamma': '#AFC4E8',
            'line_main': '#2F3A40', 'line_aux': '#AEB9BF', 'bg': '#F7FAFA', 'track': '#E8EEEE',
            'section': '#E6A66F', 'section_edge': '#9C5F2D', 'cut': '#BFD4E6',
        }
        SNAP_COLORS = ['#5B8E7D', '#B56B45', '#7C6DAF', '#3C7A9E', '#A0754E']  # 记录快照循环色
        
        
        def cube_section(n):
            """平面 n·p=0 切立方体 [-0.5,0.5]³ 的截面多边形。
            返回 (有序 3D 顶点, 平面内 2D 顶点, 面积, 平面正交基 u/v)。"""
            n = np.asarray(n, float)
            if np.linalg.norm(n) < 1e-6:
                n = np.array([0.0, 0.0, 1.0])
            c = 0.5
            pts = []
            for i in range(3):                                  # 沿 i 轴变化的棱
                for fixed in itertools.product([-c, c], repeat=2):
                    v1 = np.zeros(3); v2 = np.zeros(3)
                    others = [a for a in range(3) if a != i]
                    v1[others[0]], v1[others[1]] = fixed
                    v2[others[0]], v2[others[1]] = fixed
                    v1[i] = -c; v2[i] = c
                    d = n @ (v2 - v1)
                    if abs(d) < 1e-12:
                        continue
                    t = -n @ v1 / d
                    if 0.0 <= t <= 1.0:
                        pts.append(v1 + t * (v2 - v1))
            if len(pts) < 3:
                return np.zeros((0, 3)), np.zeros((0, 2)), 0.0, np.eye(3)[:2]
            pts = np.array(pts)
            n_hat = n / np.linalg.norm(n)
            ref = np.array([1.0, 0.0, 0.0]) if abs(n_hat[0]) < 0.9 else np.array([0.0, 1.0, 0.0])
            u = np.cross(n_hat, ref); u /= np.linalg.norm(u)
            v = np.cross(n_hat, u)
            pts2d = pts @ np.array([u, v]).T
            cen = pts2d.mean(axis=0)
            order = np.argsort(np.arctan2(pts2d[:, 1] - cen[1], pts2d[:, 0] - cen[0]))
            pts3d, pts2d = pts[order], pts2d[order]
            x, y = pts2d[:, 0], pts2d[:, 1]
            area = 0.5 * abs(np.dot(x, np.roll(y, -1)) - np.dot(y, np.roll(x, -1)))
            return pts3d, pts2d, area, np.array([u, v])
        
        
        # ============ 六律：控件仪式抽函数 ============
        def style_control_axes(ax):
            ax.set_facecolor(MORANDI['bg'])
            for sp in ax.spines.values():
                sp.set_visible(False)
            ax.set_xticks([]); ax.set_yticks([])
        
        
        def make_slider(rect, label, lo, hi, init):
            ax = fig.add_axes(rect); style_control_axes(ax)
            s = Slider(ax, label, lo, hi, valinit=init, valfmt='%.2f',
                       color=MORANDI['plane_beta'], track_color=MORANDI['track'], initcolor='none',
                       handle_style={'facecolor': MORANDI['line_main'], 'edgecolor': 'white', 'size': 8})
            s.label.set_color(MORANDI['line_main']); s.valtext.set_color(MORANDI['line_main'])
            return s
        
        
        def make_button(rect, label):
            ax = fig.add_axes(rect); style_control_axes(ax)
            for sp in ax.spines.values():
                sp.set_color(MORANDI['line_aux']); sp.set_linewidth(0.8)
            btn = Button(ax, label, color='white', hovercolor='#F1F6F6')
            btn.label.set_color(MORANDI['line_main'])
            return btn
        
        
        def make_checkbutton(rect, labels, actives):
            ax = fig.add_axes(rect); style_control_axes(ax)
            return CheckButtons(ax, labels, actives=actives,
                label_props={'color': [MORANDI['line_main']]},
                frame_props={'edgecolor': [MORANDI['line_aux']], 'facecolor': ['white'],
                             'linewidth': [0.8], 's': [36]},
                check_props={'facecolor': [MORANDI['line_main']], 'linewidth': [1.2], 's': [36]})

        fig = plt.figure(figsize=(11, 5.5), facecolor=MORANDI['bg'])

        ax1 = fig.add_axes([0.02, 0.20, 0.48, 0.72], projection='3d', proj_type='ortho')
        ax1.set_axis_off(); ax1.set_facecolor(MORANDI['bg']); ax1.view_init(elev=20, azim=-45)
        ax1.set_xlim(-0.65, 0.65); ax1.set_ylim(-0.65, 0.65); ax1.set_zlim(-0.65, 0.65)
        ax1.text2D(0.5, 0.95, "自由平面切割立方体", transform=ax1.transAxes,

        ax2.spines['left'].set_color(MORANDI['line_aux']); ax2.spines['bottom'].set_color(MORANDI['line_aux'])
        ax2.set_title("截面真形", pad=12, fontsize=13, color=MORANDI['line_main'])
        ax2.set_xlim(-0.9, 0.9); ax2.set_ylim(-0.9, 0.9); ax2.set_aspect('equal')
        ax2.set_xticks([]); ax2.set_yticks([])
        
        # 立方体线框
        cube_edges = []
        c = 0.5
        for i in range(3):
            for fixed in itertools.product([-c, c], repeat=2):
                p1 = np.zeros(3); p2 = np.zeros(3)
                others = [a for a in range(3) if a != i]
                p1[others[0]], p1[others[1]] = fixed
                p2[others[0]], p2[others[1]] = fixed
                p1[i] = -c; p2[i] = c
                cube_edges.append(ax1.plot([p1[0], p2[0]], [p1[1], p2[1]], [p1[2], p2[2]],
                             color=MORANDI['line_aux'], lw=0.8, ls='--')[0])
        
        # 初始截面 + 切割平面

        pts3d0, pts2d0, area0, uv0 = cube_section(init_n)

                                       -0.8*uv0[0]-0.8*uv0[1], 0.8*uv0[0]-0.8*uv0[1]]],

            edgecolors=MORANDI['section_edge'], linewidths=1.6, alpha=0.7)

        def update(_val=None):
            n = (s_nx.val, s_ny.val, s_nz.val)
            pts3d, pts2d, area, uv = cube_section(n)
            if len(pts3d) >= 3:
                section3d.set_verts([pts3d])
                section2d.set_xy(pts2d)
                # 切割平面随法向旋转

                                      -0.8*uv[0]-0.8*uv[1], 0.8*uv[0]-0.8*uv[1]]])
            area_text.set_text(f"面积 S = {area:.4f}")
            fig.canvas.draw_idle()
        
        
        def toggle_visibility(_label):
            visible = chk.get_status()[0]
            for edge in cube_edges:
                edge.set_visible(visible)
            fig.canvas.draw_idle()
        
        
        def reveal(_e):
            if formula_text.get_text() == "":
                formula_text.set_text(r"$A_{max}=\sqrt{2}\approx1.414$ 当 $n\parallel(1,1,0)$；$A_{min}=1$ 当 $n\parallel(1,0,0)$")
                btn_reveal.label.set_text('隐藏')
            else:
                formula_text.set_text("")
                btn_reveal.label.set_text('揭示')
            fig.canvas.draw_idle()
        
        
        def record(_e):
            n = (s_nx.val, s_ny.val, s_nz.val)
            pts3d, pts2d, area, _ = cube_section(n)
            if len(pts3d) < 3:
                return
            color = SNAP_COLORS[snapshot_idx[0] % len(SNAP_COLORS)]   # 循环色
            snapshot_idx[0] += 1
            ghost3d = Poly3DCollection([pts3d], facecolors=color, edgecolors=color,

            if len(snapshots) > 5:                                    # 最多留 5 组
                g3, g2 = snapshots.pop(0)
                g3.remove(); g2.remove()
            fig.canvas.draw_idle()
        
        
        def clear_snapshots(_e=None):
            for g3, g2 in snapshots:
                g3.remove(); g2.remove()
            snapshots.clear()
            snapshot_idx[0] = 0
            fig.canvas.draw_idle()
        
        
        def reset(_e):
            clear_snapshots()
            formula_text.set_text("")
            btn_reveal.label.set_text('揭示')
            s_nx.reset(); s_ny.reset(); s_nz.reset()   # 触发 update
        
        
        for s in (s_nx, s_ny, s_nz):
            s.on_changed(update)

        btn_reveal.on_clicked(reveal)
        btn_record.on_clicked(record)
        btn_clear.on_clicked(clear_snapshots)
        
        update()
        plt.show()
